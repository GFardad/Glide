--- 11. enforcer script content ---
#!/bin/bash
set -euo pipefail

CONFIG="/home/gfardad/.config/sunshine/sunshine.conf"
TARGET="HDMI-A-4"

# Always force virtual monitor target - never fall back to physical displays
CURRENT=$(grep '^output_name = ' "$CONFIG" | cut -d= -f2 | tr -d ' ' || echo "")

if [ "$CURRENT" != "$TARGET" ]; then
    sed -i "s/^output_name = .*/output_name = $TARGET/" "$CONFIG"
    if systemctl --user is-active --quiet sunshine.service; then
        systemctl --user restart sunshine.service
    fi
fi
